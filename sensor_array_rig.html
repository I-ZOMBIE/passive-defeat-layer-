# fiber_logger.py // RPi
import RPi.GPIO as GPIO
import time, json
 
# Mech poles + fiber channels
POLES = [4,5,6,7,8,9,10,11]
FIBER = [17,18,19,20]
events = []
 
def on_mech(ch):
  t = time.time_ns()
  events.append(
    {"type":"MECH","pin":ch,"t":t})
 
def on_flash(ch):
  t = time.time_ns()
  events.append(
    {"type":"FLASH","ch":ch,"t":t})
  print(f"💡FLASH ch{ch}@{t}")
 
for p in POLES:
  GPIO.setup(p,GPIO.IN)
  GPIO.add_event_detect(p,
    GPIO.RISING,on_mech,1)
for f in FIBER:
  GPIO.setup(f,GPIO.IN)
  GPIO.add_event_detect(f,
    GPIO.RISING,on_flash,1)
 
# Saves JSON on exit
GPIO.wait_for_edge(
  POLES[0],GPIO.RISING)
